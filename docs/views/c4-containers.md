```mermaid
flowchart TB
    broker["**Брокер сообщений**<br/>(общая шина данных компании. В неё внешние доменные сервисы публикуют свои события при изменении данных)"]

    subgraph ns["Личный кабинет"]
        web["**Веб-приложение (SPA)**<br/>[React]"]

        bff["**Бэкенд-сервис BFF**<br/>Легковесный API. Выполняет один быстрый запрос в локальную базу данных личного кабинета."]
        agregator["**Сервис-Агрегатор**<br/>Фоновый процесс, который слушает брокер сообщений и обновляет агрегированные данные пользователей."]
    end

    web --> |"запрашивает данные дашборда"| bff
    bff --> |"читает данные"| db
    agregator --> |"Обновляет агрегированные данные"| db
    agregator --> |"Подписывается на события"| broker
```
